  const usage = await recordUsage(job, model, budget, ai, 'SPECIALIST_EXECUTION');
  const structured = safeJson(ai.output, 'SPECIALIST_OUTPUT_JSON_INVALID');
  const invalidSources = structured.confirmed_facts.filter(item => !contextKeys.includes(String(item.source_context_key)));
  if (invalidSources.length) throw new Error('SPECIALIST_SOURCE_REFERENCE_INVALID');
  await step(job, 'SPECIALIST_EXECUTION', 'COMPLETED', {context_snapshot_hash:snapshotHash,usage,requires_human_confirmation:structured.requires_human_confirmation});
  return completeJob(job, {
    provider:model.provider,model:model.model_name,response_id:ai.responseId,
    output_text:structured.deliverable,structured,usage:ai.usage,estimated_cost_thb:usage.estimatedCostThb,
    capability_key:capability.capability_key,capability_version:capability.version,
    context_keys:contextKeys,context_snapshot:snapshotItems,context_snapshot_hash:snapshotHash,
    external_write_performed:false,quality_gate:{status:'SPECIALIST_STRUCTURED',source_validation:'PASSED'}
  });
}

async function reviewDraft(job, draft, children, round, contextSnapshot, policy) {
  const reviewJob = {...job,owner_agent_id:'audit'};
  const model = await modelFor(reviewJob, 'reviewer');
  const {budget} = await validateRuntime(model);
  const instructions = [
    'You are the independent Audit and final quality gate for Unity Global AI Office.',
    'Review the PMO draft against the Managing Director request, specialist evidence and authoritative governance context.',
    `A PASS requires score >= ${policy.min_pass_score}, no unsupported factual claim, no hidden failed work, no material conflict, explicit risk disclosure and direct instruction compliance.`,
    'Use REVISE when defects are fixable. Use BLOCK for fabricated facts, unresolved critical conflict, unsafe recommendation or missing essential evidence.',
    'Do not rewrite the answer. Return only the structured review.',
    `Original request: ${job.title}`,
    `PMO draft: ${JSON.stringify(draft)}`,
    `Specialist evidence: ${JSON.stringify(children)}`,
    `Governance context snapshot: ${JSON.stringify(contextSnapshot)}`
  ].join('\n');
  const ai = await callOpenAI(model, instructions, 'Perform the independent final review.', {schema:REVIEW_SCHEMA,schemaName:'pmo_quality_review',schemaDescription:'Independent PMO quality gate'});
  const usage = await recordUsage(job, model, budget, ai, `PMO_REVIEW_ROUND_${round}`);
  const review = safeJson(ai.output, 'PMO_REVIEW_JSON_INVALID');
  if (review.verdict === 'PASS' && Number(review.score) < policy.min_pass_score) review.verdict = 'REVISE';
  if (policy.block_on_unsupported_claim && review.unsupported_claims.length > 0 && review.verdict === 'PASS') review.verdict = 'REVISE';
  if (policy.block_on_critical_conflict && review.defects.some(item => item.severity === 'CRITICAL')) review.verdict = 'BLOCK';
  await createReview(job, round, model, review, ai, usage, contextSnapshot);
  await event('PMO_QUALITY_REVIEW', `PMO 품질검수 ${round}: ${review.verdict} ${review.score}`, 'audit', job.id, {round,verdict:review.verdict,score:review.score,defects:review.defects.length});
  return {review,model,ai,usage};
}
