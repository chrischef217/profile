  await event('JOB_FAILED', `AI 실행 실패: ${job.title}`, job.owner_agent_id, job.id, {error:message.slice(0,300),stage:STAGE});
  return {ok:false,status:'FAILED',job_id:job.id,error:message.slice(0,300)};
}

async function completeJob(job, result) {
  const completed = new Date().toISOString();
  await releaseJob(job, {status:'COMPLETED',phase:'DONE',result,error_message:null,completed_at:completed});
  await setAgent(job.owner_agent_id, 'IDLE', null, '다음 업무 대기');
  await step(job, 'JOB_TERMINAL', 'COMPLETED', {quality_gate:result.quality_gate || null,provider:result.provider,model:result.model});
  await event('JOB_COMPLETED', `AI 결과 생성 완료: ${job.title}`, job.owner_agent_id, job.id, {
    provider:result.provider,model:result.model,response_id:result.response_id,job_type:job.job_type,phase:job.phase,quality_gate:result.quality_gate || null
  });
  return {ok:true,status:'COMPLETED',job_id:job.id,provider:result.provider,model:result.model,quality_gate:result.quality_gate || null};
}

async function qualityPolicy() {
  const policy = await one('ai_office_quality_policy', 'select=*&id=eq.primary');
  return {
    enabled:policy?.enabled !== false,
    min_pass_score:Number(policy?.min_pass_score || MIN_PASS_SCORE),
    max_revisions:Number.isFinite(Number(policy?.max_revisions)) ? Number(policy.max_revisions) : MAX_REVISIONS,
    block_on_unsupported_claim:policy?.block_on_unsupported_claim !== false,
    block_on_critical_conflict:policy?.block_on_critical_conflict !== false
  };
}

async function validateRuntime(model) {
  const config = await one('ai_office_execution_config', 'select=*&id=eq.primary');
  if (!config?.model_enabled) throw new Error(String(config?.disabled_reason || 'MODEL_NOT_ENABLED'));
  if (String(model.provider) !== 'OPENAI') throw new Error('PROVIDER_NOT_IMPLEMENTED');
  if (!OPENAI_KEY) throw new Error('OPENAI_API_KEY_NOT_CONFIGURED');
  const budget = await budgetState(model);
  enforceBudget(budget);
  return {config,budget};
}
